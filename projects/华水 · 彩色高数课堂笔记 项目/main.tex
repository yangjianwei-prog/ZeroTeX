\section{极限与连续}

\begin{defbox}[函数极限]
设 $f$ 在 $x_0$ 的去心邻域内有定义。若存在常数 $A$，对任意
$\varepsilon>0$ 都存在 $\delta>0$，使得当 $0<\lvert x-x_0\rvert<\delta$ 时
$\lvert f(x)-A\rvert<\varepsilon$，则称
\begin{equation}\label{eq:limit}
  \lim_{x\to x_0}f(x)=A .
\end{equation}
\end{defbox}

式 \eqref{eq:limit} 是全部微分学的起点；经典处理方式可参见~\cite{Evans,Rudin}。

\begin{tipbox}[记忆口诀]
「$\varepsilon$ 先来，$\delta$ 后到」：先给定精度，再找半径。
\end{tipbox}

\section{导数与中值定理}

\begin{thmbox}[拉格朗日中值定理]
若 $f\in C[a,b]$ 且在 $(a,b)$ 可导，则存在 $\xi\in(a,b)$ 使
\begin{equation}\label{eq:mvt}
  f'(\xi)=\frac{f(b)-f(a)}{b-a} .
\end{equation}
\end{thmbox}

由 \eqref{eq:mvt} 立刻得到单调性判别法与误差估计，详见~\cite{Rudin}。

\begin{sumbox}[本节小结]
极限刻画局部行为，导数刻画变化率，中值定理把二者联系起来。
\end{sumbox}
