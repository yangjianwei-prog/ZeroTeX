\begin{frame}{罗尔定理}
\begin{theobox}{罗尔 (Rolle) 定理}
若 $f\in C[a,b]$、在 $(a,b)$ 可导且 $f(a)=f(b)$，则存在
$\xi\in(a,b)$ 使
\begin{equation}\label{eq:rolle}
  f'(\xi)=0 .
\end{equation}
\end{theobox}

式 \eqref{eq:rolle} 是拉格朗日中值定理的特例；一般形式见~\cite{Tongji,Zorich}。
\end{frame}

\begin{frame}{拉格朗日中值定理}
\begin{theobox}{Lagrange 中值定理}
在上述条件下存在 $\xi\in(a,b)$ 使
\begin{equation}\label{eq:lagrange}
  f(b)-f(a)=f'(\xi)(b-a).
\end{equation}
\end{theobox}

\begin{corobox}{推论}
若 $f'(x)\equiv0$ 于区间 $I$，则 $f$ 在 $I$ 上为常数；
由 \eqref{eq:lagrange} 立即得证，参见~\cite{Tongji}。
\end{corobox}
\end{frame}

\begin{frame}{应用举例}
\begin{exambox}{证明不等式}
当 $x>0$ 时，由 \eqref{eq:lagrange} 作用于 $\ln(1+x)$ 可得
$\frac{x}{1+x}<\ln(1+x)<x$。
\end{exambox}
