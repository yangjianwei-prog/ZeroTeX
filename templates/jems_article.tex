\begin{abstract}
We prove sharp lower bounds for the first eigenvalue of the Laplace-Beltrami operator on a class of asymptotically conical Riemannian manifolds.
\end{abstract}
\paragraph{Keywords} Laplace-Beltrami operator; eigenvalue estimate; asymptotically conical; Weyl law

\section{Introduction}
Let $(M,g)$ be a complete noncompact Riemannian manifold asymptotic to a cone over a compact cross-section $\Sigma$.
For the classical background and related results, see~\cite{Evans,Brezis}.

\begin{theorem}
If $\mathrm{Ric}_g\ge0$ outside a compact set, then $\lambda_1(M)\ge\frac{(n-2)^2}{4}\liminf_{r\to\infty}\lambda_1(\Sigma_r)$.
\end{theorem}

\begin{proof}
The estimate follows from a weighted Bochner formula and a Hardy inequality adapted to the conical end of $M$.
\end{proof}

\section{Main estimates}
\begin{equation}
-\Delta_g u=\lambda u,\qquad \int_M|\nabla_g u|^2\,\dd V_g=\lambda\int_M u^2\,\dd V_g,
\end{equation}
The bound is attained asymptotically by the model cone, hence it is sharp.
