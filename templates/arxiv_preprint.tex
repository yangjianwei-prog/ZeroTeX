\begin{abstract}
We consider the vanishing viscosity limit for a scalar conservation law
with a nonconvex flux. Under a compensated compactness framework we show
that the limit is the unique entropy solution. \emph{This template is

\end{abstract}

\section{Introduction}

Consider the parabolic approximation
\begin{equation}\label{eq:visc}
  \partial_t u^\eps + \partial_x f(u^\eps) = \eps\,\partial_{xx}u^\eps,
  \qquad u^\eps(x,0)=u_0(x),
\end{equation}
on $\R\times(0,T)$. We are interested in the limit $\eps\to0^+$.

\begin{theorem}
Assume $u_0\in L^\infty(\R)$ and $f\in C^2(\R)$. Then
$u^\eps\to u$ in $L^1_{\mathrm{loc}}(\R\times(0,T))$, where $u$ is the
unique Kru{\v z}kov entropy solution of
$\partial_t u+\partial_x f(u)=0$.
\end{theorem}

\begin{proof}
The uniform $L^\infty$ bound and the entropy production estimate give
$\eps\,\partial_x u^\eps\to0$ in the sense of measures. Compensated
compactness (Tartar--Murat) then yields strong convergence.
\end{proof}
