\begin{abstract}
We analyze an accelerated proximal gradient scheme for minimizing the sum of a smooth convex function and a nonsmooth convex regularizer.
\end{abstract}
\paragraph{Keywords} proximal gradient; acceleration; convex optimization; convergence rate

\section{Introduction}
We consider $\min_{x\in\R^d} F(x)=f(x)+g(x)$ where $f$ is $L$-smooth and convex and $g$ is proper closed convex with an efficiently computable proximity operator.
For the classical background and related results, see~\cite{Evans,Brezis}.

\begin{theorem}
The iterates generated by the proposed scheme satisfy $F(x_k)-F(x^\ast)\le \frac{2L\|x_0-x^\ast\|^2}{(k+1)^2}$ for all $k\ge1$.
\end{theorem}
