\begin{abstract}
We prove existence and uniqueness of renormalized solutions to a
nonlocal diffusion equation with a bounded Radon measure as right-hand
side. The proof relies on a comparison principle and truncation
techniques.
\keywords{Nonlocal diffusion \and Measure data \and Renormalized solution}
\PACS{02.30.Jr \and 02.60.Lj}
\subclass{35R11 \and 35D99}
\end{abstract}

\section{Introduction and main result}

Let $\Om\subset\R^n$ be a bounded open set and let $\mu$ be a bounded
Radon measure on $\Om$. We consider
\begin{equation}\label{eq:nl}
  (-\Delta)^s u = \mu \ \text{ in } \Om, \qquad u = 0 \ \text{ in } \R^n\setminus\Om,
\end{equation}
with $s\in(0,1)$.

\begin{theorem}
For every bounded Radon measure $\mu$ there exists a unique
renormalized solution $u$ of \eqref{eq:nl}, and
$u\in L^q(\R^n)$ for all $q<n/(n-2s)$.
\end{theorem}

\section{Proof}
Approximate $\mu$ by smooth functions, derive uniform estimates via
the fractional Sobolev embedding, and pass to the limit using
compactness.
